━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📋 1分で！PC のIPアドレスを取得
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

【Windows の場合】
  1. Windows キー + R
  2. 「powershell」と入力 → Enter
  3. コピー＆ペースト：
     
     ipconfig | find "IPv4"
     
  4. Enter を押す
  5. 「IPv4 アドレス」の数字をメモ

【Mac の場合】
  1. Spotlight（cmd + Space）で「ターミナル」検索
  2. ターミナルを開く
  3. コピー＆ペースト：
     
     ifconfig | grep "inet " | grep -v 127.0.0.1
     
  4. Enter を押す
  5. 「inet」の後の数字をメモ

【メモするIPアドレス】（このように見える）
  192.168.1.100
  ※ 上記は例です。自分のものをメモしてください


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📋 PC でサーバーを起動（ターミナル/PowerShell）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

コマンドをコピー＆ペースト：

  cd /workspaces/anki-system
  python3 -m http.server 8000

成功時の表示：
  Serving HTTP on 0.0.0.0 port 8000 ...

⚠️  ここではターミナルを閉じないでください！


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📋 Android + Brave でアクセス
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

手順：
  1. Brave ブラウザを開く
  2. URL 入力欄をタップ
  3. 以下を入力（自分のIPを入れる）：
     
     http://192.168.1.100:8000
     
     ※ 自分のIPアドレスに置き換え
  4. 虫眼鏡（検索）をタップ
  5. 3-5秒待つ
  6. ✅ アプリが表示される！

確認：
  □ 暗いグラデーション背景
  □ 「弁理士試験 暗記アプリ」というタイトル
  □ 「学習」と「📋 データ管理」タブが見える


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
📋 PWA をホーム画面に追加
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

手順：
  1. Brave でアプリが開いている状態を保つ
  2. 画面右下の「⋮」（3点メニュー）をタップ
  3. 「アプリをインストール」をタップ
     ※ 見つからなければ下にスクロール
  4. 「インストール」ボタンをタップ
  5. ✅ 完了！

確認：
  □ ホーム画面に「弁理士暗記」アイコン（紫色）が追加された
  □ アイコンをタップで全画面でアプリが起動


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
⚠️  トラブル シューティング
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ 「接続できません」
  → PC と スマホが同じ WiFi に接続しているか確認
  → IPアドレスが正しいか確認
  → http:// と :8000 が入っているか確認

❌ 「ページが見つかりません」
  → PC のターミナルで Serving HTTP と表示されているか確認
  → サーバーが起動しているか確認

❌ 「アプリをインストール」が表示されない
  → Brave でページを再読み込み（下にスワイプ）
  → 3-5秒待つ
  → もう一度メニュー（⋮）を確認
